--[[
	FLY MENU v2  -  fly, swim, float (air-walk), noclip, infinite jump, walk/jump speed,
	click teleport, fly trail, live stats, themes + full colour customization.

	Put this in a LocalScript inside StarterPlayer > StarterPlayerScripts

	FLY (default key E): your character faces exactly where the camera looks.
	  W A S D = move toward where you look, Space = up, Left Ctrl = down.
	FLOAT: you stand on an invisible platform, so you WALK in the air (real walk animation).
	  Space / Left Ctrl move the platform up / down.
	CTRL + CLICK TELEPORT: turn it on in the Player tab, then hold Left Ctrl and click.

	BUILD tab: "Add 1 Block" with shape, size, material, colour and placement options.
	  (needs the FlyMenuBuildServer script in ServerScriptService so blocks are real for everyone)

	FIRST LAUNCH: pick PC or Mobile.
	  PC     -> choose a key to minimize / reopen the menu (and use the keyboard shortcuts).
	  Mobile -> a draggable airplane button opens / closes the menu, plus side buttons
	            for fly, up, down and add-block.

	ADMIN tab: a command console (press ; on PC).  Try:  ;cmds   ;fly 80   ;noclip
	           ;build ball 6 neon red   ;theme ocean   ;tp <player>
	           Commands only affect YOU (this is a client script).

	FUN tab: spin, bouncy landings, rainbow body, disco lighting, auras, gravity presets,
	         launch, random teleport, emotes, confetti, party mode, dice and coin flip.

	AIM / ESP / HITBOX tabs (for "Exploiter Frenzy" style games): aimbot, silent aim, test blaster,
	         triggerbot, ESP (boxes, names, health, tracers, chams), server-checked hitbox expander.
	         Needs the FlyMenuCombatServer script in ServerScriptService for shooting + hitboxes.

	Tabs: Fly | Player | Build | Fun | Aim | ESP | Hitbox | Admin | Theme | Config  (swipe the tab bar).
	When the menu is minimized a small floating button appears so you can always bring it back.
]]

local THEME = { good = C(80, 220, 130), bad = C(240, 85, 95) }
local registry = {} -- {instance, property, colorFunction}

local function computeDerived()
	THEME.accentHover = THEME.accent:Lerp(Color3.new(1, 1, 1), 0.18)
	THEME.boxHover = THEME.box:Lerp(Color3.new(1, 1, 1), 0.1)
end

local function loadPreset(p)
	for _, k in ipairs({ "bg", "bar", "box", "accent", "text", "subtext" }) do
		THEME[k] = p[k]
	end
	computeDerived()
end
loadPreset(PRESETS[1])

local onThemeApplied -- set later (trail color, etc.)

local function bind(inst, prop, keyOrFn)
	local fn = type(keyOrFn) == "function" and keyOrFn or function() return THEME[keyOrFn] end
	-- remember whether this depends on the accent colour (lets the rainbow update only those)
	local dependsOnAccent = type(keyOrFn) == "function" or keyOrFn == "accent" or keyOrFn == "accentHover"
	table.insert(registry, { inst, prop, fn, dependsOnAccent })
	inst[prop] = fn()
end

local TweenService_ = TweenService
local function tween(obj, time, props, style, dir)
	local t = TweenService_:Create(
		obj,
		TweenInfo.new(time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	t:Play()
	return t
end

-- instant: set directly (no tween).  onlyAccent: skip everything that doesn't use the accent colour
local function applyTheme(instant, onlyAccent)
	computeDerived()
	for i = #registry, 1, -1 do
		local e = registry[i]
		local inst = e[1]
		if inst.Parent == nil then
			table.remove(registry, i) -- destroyed
		elseif not onlyAccent or e[4] then
			if instant then
				inst[e[2]] = e[3]()
			else
				tween(inst, 0.3, { [e[2]] = e[3]() })
			end
		end
	end
	if onThemeApplied then onThemeApplied() end
end

local getCharacterParts
do
	local cChar, cHum, cRoot -- cached, so the per-frame loops don't search the character every frame
	function getCharacterParts()
		local char = player.Character
		if not char then return end
		if char ~= cChar or not cHum or not cHum.Parent or not cRoot or not cRoot.Parent then
			cChar = char
			cHum = char:FindFirstChildOfClass("Humanoid")
			cRoot = char:FindFirstChild("HumanoidRootPart")
		end
		return cHum, cRoot
	end
end

local function feetOffset(hum, root)
	local extra = hum.HipHeight
	if hum.RigType == Enum.HumanoidRigType.R6 then extra = 2 end
	return root.Size.Y / 2 + extra
end

local function updateTrailColor()
	if trail then
		trail.Color = ColorSequence.new(THEME.accent, THEME.accent:Lerp(Color3.new(1, 1, 1), 0.5))
	end
end
onThemeApplied = updateTrailColor

local function destroyTrail()
	if trail then trail:Destroy() trail = nil end
	if trailA0 then trailA0:Destroy() trailA0 = nil end
	if trailA1 then trailA1:Destroy() trailA1 = nil end
end

local function syncTrail()
	if mode and trailOn then
		if trail then return end
		local _, root = getCharacterParts()
		if not root then return end
		trailA0 = Instance.new("Attachment")
		trailA0.Position = Vector3.new(0, 1, 0)
		trailA0.Parent = root
		trailA1 = Instance.new("Attachment")
		trailA1.Position = Vector3.new(0, -1, 0)
		trailA1.Parent = root

		trail = Instance.new("Trail")
		trail.Attachment0 = trailA0
		trail.Attachment1 = trailA1
		trail.Lifetime = 0.55
		trail.MinLength = 0.05
		trail.LightEmission = 0.7
		trail.FaceCamera = true
		trail.WidthScale = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 1),
			NumberSequenceKeypoint.new(1, 0),
		})
		trail.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.2),
			NumberSequenceKeypoint.new(1, 1),
		})
		trail.Parent = root
		updateTrailColor()
	else
		destroyTrail()
	end
end

---------------------------------------------------------------------
-- FLOAT  (invisible platform -> you really WALK in the air)
---------------------------------------------------------------------
local function stopFloat()
	if floatConn then floatConn:Disconnect() floatConn = nil end
	if floatLV then floatLV:Destroy() floatLV = nil end
	if floatAttachment then floatAttachment:Destroy() floatAttachment = nil end
	if floatPlatform then floatPlatform:Destroy() floatPlatform = nil end
end

local function startFloat()
	if floatPlatform then return end
	local hum, root = getCharacterParts()
	if not hum or not root or hum.Health <= 0 then return end

	floatY = root.Position.Y - feetOffset(hum, root) -- height of your feet
	floatVel = 0

	-- stop falling right where you are
	local v = root.AssemblyLinearVelocity
	root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)

	floatPlatform = Instance.new("Part")
	floatPlatform.Name = "FloatPlatform"
	floatPlatform.Size = Vector3.new(14, 1, 14)
	floatPlatform.Anchored = true
	floatPlatform.CanCollide = true
	floatPlatform.CanTouch = false
	floatPlatform.Transparency = 1
	floatPlatform.CFrame = CFrame.new(root.Position.X, floatY - 0.55, root.Position.Z)
	floatPlatform.Parent = workspace

	-- only used while moving up/down so you ride the platform
	floatAttachment = Instance.new("Attachment")
	floatAttachment.Parent = root
	floatLV = Instance.new("LinearVelocity")
	floatLV.Attachment0 = floatAttachment
	floatLV.RelativeTo = Enum.ActuatorRelativeTo.World
	floatLV.VelocityConstraintMode = Enum.VelocityConstraintMode.Line
	floatLV.LineDirection = Vector3.yAxis
	floatLV.MaxForce = math.huge
	floatLV.LineVelocity = 0
	floatLV.Enabled = false
	floatLV.Parent = root

	floatConn = RunService.Heartbeat:Connect(function(dt)
		local hum2, root2 = getCharacterParts()
		if not floatPlatform or not hum2 or not root2 then return end
		dt = math.min(dt, 0.1)

		local target = 0
		if not UserInputService:GetFocusedTextBox() then
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then target += CFG.FLOAT_VERT_SPEED end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then target -= CFG.FLOAT_VERT_SPEED end
		end
		if touch.up then target += CFG.FLOAT_VERT_SPEED end
		if touch.down then target -= CFG.FLOAT_VERT_SPEED end
		floatVel += (target - floatVel) * (1 - math.exp(-12 * dt))
		if target == 0 and math.abs(floatVel) < 0.3 then floatVel = 0 end

		-- don't push the platform into the ground
		if floatVel < 0 then
			floatParams.FilterDescendantsInstances = { player.Character, floatPlatform }
			local p = root2.Position
			local hit = workspace:Raycast(Vector3.new(p.X, floatY, p.Z), Vector3.new(0, -2.5, 0), floatParams)
			if hit and floatY - hit.Position.Y < 0.6 then
				floatVel = 0
			end
		end

		floatY += floatVel * dt
		local p = root2.Position
		local vel = root2.AssemblyLinearVelocity
		floatPlatform.CFrame = CFrame.new(p.X + vel.X * 0.04, floatY - 0.55, p.Z + vel.Z * 0.04)

		floatLV.Enabled = floatVel ~= 0
		floatLV.LineVelocity = floatVel
	end)
end

local function syncFloat()
	if floatOn and mode == nil then
		startFloat()
	else
		stopFloat()
	end
end

local noclipConn
local noclipOriginal = {}

local function setNoclip(on)
	noclipOn = on
	if noclipConn then noclipConn:Disconnect() noclipConn = nil end
	if on then
		noclipConn = RunService.Stepped:Connect(function()
			local char = player.Character
			if not char then return end
			for _, part in ipairs(char:GetDescendants()) do
				if part:IsA("BasePart") then
					if noclipOriginal[part] == nil then noclipOriginal[part] = part.CanCollide end
					part.CanCollide = false
				end
			end
		end)
	else
		for part, was in pairs(noclipOriginal) do
			if part.Parent then part.CanCollide = was end
		end
		noclipOriginal = {}
	end
end

-- TELEPORT

local function teleportTo(pos)
	local hum, root = getCharacterParts()
	if not root then return end
	local rot = root.CFrame - root.CFrame.Position
	local offset = hum and feetOffset(hum, root) or 3
	root.CFrame = CFrame.new(pos + Vector3.new(0, offset + 0.5, 0)) * rot
	root.AssemblyLinearVelocity = Vector3.zero
	if floatPlatform then floatY = pos.Y + 0.5 end
end

---------------------------------------------------------------------
-- SWIM ANIMATIONS (default R15 swim animations)
---------------------------------------------------------------------
local function stopSwimAnims()
	for _, t in pairs(swimTracks) do
		pcall(function()
			if t.IsPlaying then t:Stop(0.2) end
		end)
	end
	swimTracks = {}
end

local function playSwim(hum, moving)
	if hum.RigType ~= Enum.HumanoidRigType.R15 then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then return end

	if not swimTracks.move then
		pcall(function()
			local a = Instance.new("Animation")
			a.AnimationId = "rbxassetid://507784897"
			local b = Instance.new("Animation")
			b.AnimationId = "rbxassetid://507785072"
			swimTracks.move = animator:LoadAnimation(a)
			swimTracks.idle = animator:LoadAnimation(b)
			for _, t in pairs(swimTracks) do
				t.Looped = true
				t.Priority = Enum.AnimationPriority.Action
			end
		end)
	end

	local want = moving and swimTracks.move or swimTracks.idle
	local other = moving and swimTracks.idle or swimTracks.move
	if other and other.IsPlaying then other:Stop(0.25) end
	if want and not want.IsPlaying then want:Play(0.25) end
end

-- FLY / SWIM ENGINE
---------------------------------------------------------------------
local function stopFlying(respawned)
	if not mode then return end
	mode = nil

	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
	if linearVelocity then linearVelocity:Destroy() linearVelocity = nil end
	if attachment then attachment:Destroy() attachment = nil end
	destroyTrail()

	local carried = velocity
	velocity = Vector3.zero

	if fovTween then fovTween:Cancel() end
	fovTween = TweenService:Create(
		camera,
		TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{ FieldOfView = defaultFov }
	)
	fovTween:Play()
	currentFov = defaultFov

	stopSwimAnims()

	if not respawned then
		local hum, root = getCharacterParts()
		if root then
			root.CFrame = CFrame.lookAt(root.Position, root.Position + lastFlat)
			if carried.Magnitude > 60 then carried = carried.Unit * 60 end
			root.AssemblyLinearVelocity = carried
			root.AssemblyAngularVelocity = Vector3.zero
		end
		if hum then
			hum.PlatformStand = false
			hum.AutoRotate = true
			hum:ChangeState(Enum.HumanoidStateType.GettingUp)
		end
	end

	syncFloat()
	if updateStatus then updateStatus() end
	if refreshUI then refreshUI() end
end

local function startFlying(newMode)
	-- switching between fly <-> swim without stopping
	if mode then
		mode = newMode
		if mode == "fly" then stopSwimAnims() end
		if updateStatus then updateStatus() end
		if refreshUI then refreshUI() end
		return
	end

	local hum, root = getCharacterParts()
	if not hum or not root or hum.Health <= 0 then
		if refreshUI then refreshUI() end
		return
	end

	mode = newMode
	stopFloat()

	-- keep momentum (jump / fall) so takeoff is seamless
	velocity = root.AssemblyLinearVelocity
	if velocity.Magnitude > 80 then velocity = velocity.Unit * 80 end
	currentRot = root.CFrame.Rotation
	bobClock = 0

	if fovTween then fovTween:Cancel() end
	currentFov = camera.FieldOfView

	attachment = Instance.new("Attachment")
	attachment.Name = "FlyAttachment"
	attachment.Parent = root

	linearVelocity = Instance.new("LinearVelocity")
	linearVelocity.Attachment0 = attachment
	linearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
	linearVelocity.MaxForce = math.huge
	linearVelocity.VectorVelocity = velocity
	linearVelocity.Parent = root

	hum.PlatformStand = true
	hum.AutoRotate = false
	syncTrail()

	flyConnection = RunService.RenderStepped:Connect(function(dt)
		local hum2, root2 = getCharacterParts()
		if not hum2 or not root2 or not linearVelocity then return end
		dt = math.min(dt, 0.1)

		camera = workspace.CurrentCamera
		local cf = camera.CFrame
		local look, right = cf.LookVector, cf.RightVector
		local speed = (mode == "swim") and flySpeed * CFG.SWIM_RATIO or flySpeed

		-- input -> desired direction (full 3D, follows the camera)
		local dir = Vector3.zero
		if not UserInputService:GetFocusedTextBox() then
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += look end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= look end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += right end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= right end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
		end

		-- phones: the jump button flies you up
		if not UserInputService.KeyboardEnabled and hum2.Jump then
			dir += Vector3.yAxis
		end
		if touch.up then dir += Vector3.yAxis end
		if touch.down then dir -= Vector3.yAxis end

		-- mobile / gamepad: thumbstick pushes you toward where the camera looks
		local md = hum2.MoveDirection
		if dir.Magnitude < 0.01 and md.Magnitude > 0 then
			local flatLook = Vector3.new(look.X, 0, look.Z)
			local flatRight = Vector3.new(right.X, 0, right.Z)
			if flatLook.Magnitude > 0.001 and flatRight.Magnitude > 0.001 then
				dir = look * md:Dot(flatLook.Unit) + right * md:Dot(flatRight.Unit)
			end
		end

		local hasInput = dir.Magnitude > 0.01

		-- velocity: quick to speed up, a touch of glide on release
		local target
		if hasInput then
			target = dir.Unit * speed
		else
			bobClock += dt
			target = Vector3.new(0, math.sin(bobClock * 2) * 1.2, 0)
		end
		local rate = hasInput and getAcceleration() or getAcceleration() * 0.6
		velocity = velocity:Lerp(target, 1 - math.exp(-rate * dt))
		linearVelocity.VectorVelocity = velocity

		-- orientation: character faces exactly where the camera looks
		local flat = Vector3.new(look.X, 0, look.Z)
		if flat.Magnitude > 0.001 then
			flat = flat.Unit
			lastFlat = flat
		else
			flat = lastFlat
		end
		local pitch = math.asin(math.clamp(look.Y, -1, 1))

		local speedRef = math.max(speed, 1)
		local sideRatio = math.clamp(velocity:Dot(right) / speedRef, -1, 1)
		local roll = -sideRatio * math.rad(CFG.BANK_AMOUNT)

		local lookPose = CFrame.lookAt(Vector3.zero, flat)
			* CFrame.Angles(pitch, 0, 0)
			* CFrame.Angles(0, 0, roll)

		local targetRot = lookPose
		if supermanPose or mode == "swim" then
			local superPose = lookPose * CFrame.Angles(-math.pi / 2, 0, 0)
			local fwd = math.clamp(velocity:Dot(look) / speedRef, 0, 1)
			local blend = fwd * fwd * (3 - 2 * fwd)
			targetRot = lookPose:Lerp(superPose, blend)
		end

		local rotAlpha = 1 - math.exp(-(10 + responsiveness * 1.5) * dt)
		currentRot = currentRot:Lerp(targetRot, rotAlpha)

		root2.CFrame = CFrame.new(root2.Position) * currentRot
		root2.AssemblyAngularVelocity = Vector3.zero

		if mode == "swim" then
			playSwim(hum2, hasInput)
		end

		-- speed feel: field of view widens as you go faster
		if fovEffect then
			local fovTarget = defaultFov + math.clamp(velocity.Magnitude / 120, 0, 1) * CFG.FOV_BOOST
			currentFov += (fovTarget - currentFov) * (1 - math.exp(-5 * dt))
			camera.FieldOfView = currentFov
		elseif currentFov ~= defaultFov then
			currentFov = defaultFov
			camera.FieldOfView = defaultFov
		end
	end)

	if updateStatus then updateStatus() end
	if refreshUI then refreshUI() end
end

local function toggleFly()
	if mode == "fly" then
		stopFlying()
		notify("Fly disabled", "bad")
	else
		startFlying("fly")
		notify("Fly enabled", "good")
	end
end

-- soft top-light highlight (an overlay frame, so text colours are never affected)
local function gloss(inst, radius, amount)
	local f = Instance.new("Frame")
	f.Name = "Gloss"
	f.BackgroundColor3 = Color3.new(1, 1, 1)
	f.BorderSizePixel = 0
	f.Size = UDim2.fromScale(1, 1)
	f.Active = false
	f.Parent = inst
	round(f, radius)
	local g = Instance.new("UIGradient")
	g.Rotation = 90
	g.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 1 - amount),

local function hoverable(btn, normalKey, hoverKey)
	btn.MouseEnter:Connect(function() tween(btn, 0.15, { BackgroundColor3 = THEME[hoverKey] }) end)
	btn.MouseLeave:Connect(function() tween(btn, 0.15, { BackgroundColor3 = THEME[normalKey] }) end)

	-- little "press" squish so every button feels responsive
	local sc = Instance.new("UIScale")
	sc.Parent = btn
	local function release() tween(sc, 0.2, { Scale = 1 }, Enum.EasingStyle.Back) end
	btn.MouseButton1Down:Connect(function() tween(sc, 0.08, { Scale = 0.94 }) end)
	btn.MouseButton1Up:Connect(release)
	btn.MouseLeave:Connect(release)
end

-- switch. returns set(value, silent) and get()
local function makeToggle(parent, text, y, initial, onChanged)
	label(parent, text, 16, y, 200, 30)

	local state = initial
	local pill = Instance.new("TextButton")
	pill.Size = UDim2.fromOffset(46, 24)
	pill.Position = UDim2.new(1, -66, 0, y + 3)
	pill.BorderSizePixel = 0
	pill.AutoButtonColor = false
	pill.Text = ""
	pill.Parent = parent
	round(pill, 12)
	local function pillColor() return state and THEME.accent or THEME.box end
	bind(pill, "BackgroundColor3", pillColor)

	local knob = Instance.new("Frame")
	knob.Size = UDim2.fromOffset(18, 18)
	knob.Position = state and UDim2.fromOffset(25, 3) or UDim2.fromOffset(3, 3)
	knob.BackgroundColor3 = Color3.new(1, 1, 1)
	knob.BorderSizePixel = 0
	knob.Parent = pill
	round(knob, 9)

	local function set(v, silent)
		state = v
		tween(pill, 0.18, { BackgroundColor3 = pillColor() })
		tween(knob, 0.18, { Position = v and UDim2.fromOffset(25, 3) or UDim2.fromOffset(3, 3) })
		if not silent and onChanged then onChanged(v) end
	end
	pill.MouseButton1Click:Connect(function() set(not state) end)

	return set, function() return state end
end

local function makeSlider(parent, text, y, minV, maxV, init, suffix, onChanged)
	label(parent, text, 16, y, 74, 24, Enum.Font.GothamMedium, 12, "subtext")
	local trackX = 92

	local track = Instance.new("Frame")
	track.Position = UDim2.new(0, trackX, 0, y + 9)
	track.Size = UDim2.new(1, -(trackX + 62), 0, 6)
	track.BorderSizePixel = 0
	track.Parent = parent
	round(track, 3)
	bind(track, "BackgroundColor3", "box")

	local fill = Instance.new("Frame")
	fill.BorderSizePixel = 0
	fill.Size = UDim2.fromScale(0, 1)
	fill.Parent = track
	round(fill, 3)
	bind(fill, "BackgroundColor3", "accent")

	local knob = Instance.new("Frame")
	knob.AnchorPoint = Vector2.new(0.5, 0.5)
	knob.Size = UDim2.fromOffset(14, 14)
	knob.BackgroundColor3 = Color3.new(1, 1, 1)
	knob.BorderSizePixel = 0
	knob.Parent = track
	round(knob, 7)

	local valueLbl = label(parent, "", 0, y, 48, 24, Enum.Font.GothamBold, 12, "text", Enum.TextXAlignment.Right)
	valueLbl.Position = UDim2.new(1, -64, 0, y)

	local hit = Instance.new("TextButton")
	hit.BackgroundTransparency = 1
	hit.Text = ""
	hit.Position = UDim2.new(0, trackX - 6, 0, y)
	hit.Size = UDim2.new(1, -(trackX + 50), 0, 24)
	hit.Parent = parent

	local value = init
	local function render()
		local a = (value - minV) / (maxV - minV)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		valueLbl.Text = tostring(math.floor(value + 0.5)) .. (suffix or "")
	end

	local function set(v, silent)
		value = math.clamp(v, minV, maxV)
		render()
		if not silent and onChanged then onChanged(value) end
	end
	render()

	local sliding = false
	local function fromInput(input)
		local a = math.clamp((input.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		set(minV + a * (maxV - minV))
	end
	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			fromInput(input)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			fromInput(input)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end)

	return set
end

local function localFolder()
	local f = workspace:FindFirstChild(BUILD_FOLDER)
	if not f then
		f = Instance.new("Folder")

		f.Parent = workspace

local function pruneLocal()
	for i = #localBlocks, 1, -1 do
		if not localBlocks[i].Parent then table.remove(localBlocks, i) end
	end
end

local function buildLocalPart(data)
	local part
	if data.Shape == "Wedge" then
		part = Instance.new("WedgePart")
	else
		part = Instance.new("Part")
	end
	part.Name = "Block"
	local cf = CFrame.new(data.Position) * CFrame.Angles(0, data.Yaw, 0)
	if data.Shape == "Cylinder" then
		part.Shape = Enum.PartType.Cylinder
		part.Size = Vector3.new(data.Size.Y, data.Size.X, data.Size.Z)
		cf = cf * CFrame.Angles(0, 0, math.pi / 2)
	elseif data.Shape == "Ball" then
		part.Shape = Enum.PartType.Ball
		part.Size = data.Size
	else
		part.Size = data.Size
	end
	part.CFrame = cf
	part.Color = data.Color
	part.Material = Enum.Material[data.Material]
	part.Transparency = data.Transparency
	part.Anchored = data.Anchored
	part.CanCollide = data.CanCollide
	part.TopSurface = Enum.SurfaceType.Smooth
	part.BottomSurface = Enum.SurfaceType.Smooth
	return part
end

local function callBuild(action, data)
	if buildRemote then
		local ok, res = pcall(function() return buildRemote:InvokeServer(action, data) end)
		if ok and type(res) == "table" then return res end
		return { ok = false, message = "Server did not respond" }
	end

	-- no server script found: build locally (only you can see these blocks)
	pruneLocal()
	if action == "place" then
		if #localBlocks >= BUILD_MAX_BLOCKS then
			return { ok = false, message = "Block limit reached (" .. BUILD_MAX_BLOCKS .. ")", count = #localBlocks }
		end
		local part = buildLocalPart(data)
		part.Parent = localFolder()
		table.insert(localBlocks, part)
		return { ok = true, count = #localBlocks }
	elseif action == "undo" then
		local part = table.remove(localBlocks)
		if part then part:Destroy() end
		return { ok = part ~= nil, count = #localBlocks, message = part and "" or "Nothing to undo" }
	elseif action == "clear" then
		for _, part in ipairs(localBlocks) do part:Destroy() end
		localBlocks = {}
		return { ok = true, count = 0 }
	end
	return { ok = false, message = "Unknown action" }
end

-- where the block goes + which way it faces
local function computePlacement()
	local hum, root = getCharacterParts()
	if not hum or not root then return nil end
	camera = workspace.CurrentCamera

	local sx, sy, sz = effectiveSize()
	local look = camera.CFrame.LookVector
	local flat = Vector3.new(look.X, 0, look.Z)
	if flat.Magnitude < 0.001 then flat = Vector3.zAxis else flat = flat.Unit end

	local yaw = 0
	if B.faceMe then
		yaw = math.atan2(-flat.X, -flat.Z)
		if B.snap then
			yaw = math.floor(yaw / (math.pi / 2) + 0.5) * (math.pi / 2)
		end
	end

	local pos
	if B.placeMode == "look" then
		local ignore = {}
		if player.Character then table.insert(ignore, player.Character) end
		if floatPlatform then table.insert(ignore, floatPlatform) end
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = ignore
		local hit = workspace:Raycast(camera.CFrame.Position, look * 300, params)
		if hit then
			local n = hit.Normal
			local push = (math.abs(n.X) * sx + math.abs(n.Y) * sy + math.abs(n.Z) * sz) / 2
			pos = hit.Position + n * push
		else
			pos = camera.CFrame.Position + look * B.distance
		end
	else
		local feetY = root.Position.Y - feetOffset(hum, root)
		pos = Vector3.new(
			root.Position.X + flat.X * B.distance,
			feetY + sy / 2,
			root.Position.Z + flat.Z * B.distance
		)
	end

	if B.snap then
		-- line the block's edges up with the 1-stud grid
		local hx, hz = sx / 2, sz / 2
		if math.abs(math.sin(yaw)) > 0.5 then hx, hz = hz, hx end
		local function g(v) return math.floor(v + 0.5) end
		pos = Vector3.new(
			g(pos.X - hx) + hx,
			g(pos.Y - sy / 2) + sy / 2,
			g(pos.Z - hz) + hz
		)
	end

	return pos, yaw, Vector3.new(sx, sy, sz)
end

local addBtn, undoBtn, clearBtn

function refreshBuildInfo()
	countLbl.Text = blockCount .. " / " .. BUILD_MAX_BLOCKS .. (buildRemote and "" or "   (local only)")
end

function placeBlock()
	if placing then return end
	local pos, yaw, size = computePlacement()
	if not pos then
		notify("You need a character to place blocks", "bad")
		return
	end

	if B.autoStack then
		-- pressing Add again in the same spot stacks the new block on top of the last one
		if stackBase and (pos - stackBase).Magnitude < 0.05 then
			stackLevel += 1
		else
			stackBase, stackLevel = pos, 0
		end
		pos = stackBase + Vector3.new(0, size.Y * stackLevel, 0)
	end

	placing = true
	local mat = materials[B.materialIndex]
	local res = callBuild("place", {
		Shape = B.shape,
		Size = size,
		Position = pos,
		Yaw = yaw,
		Color = B.color,
		Material = mat.name,
		Transparency = 1 - B.opacity / 100,
		Anchored = B.anchored,
		CanCollide = B.canCollide,
	})
	placing = false

	if res.count then blockCount = res.count end
	refreshBuildInfo()
	if res.ok then
		notify("Placed " .. B.shape:lower() .. "  (" .. blockCount .. " / " .. BUILD_MAX_BLOCKS .. ")", "good")
		if addBtn then flash(addBtn, THEME.good) end
	else
		if B.autoStack and stackLevel > 0 then stackLevel -= 1 end
		notify(res.message or "Could not place the block", "bad")
		if addBtn then flash(addBtn, THEME.bad) end
	end
end

local function undoBlock()
	local res = callBuild("undo")
	if res.count then blockCount = res.count end
	refreshBuildInfo()
	if res.ok then
		if stackLevel > 0 then stackLevel -= 1 end
		notify("Removed last block", "good")
	else
		notify(res.message or "Nothing to undo", "bad")
	end
end

local clearArmed = false
local function clearBlocks()
	if not clearArmed then
		-- ask for a second click so you can't wipe everything by accident
		clearArmed = true
		clearBtn.Text = "Sure?"
		tween(clearBtn, 0.15, { BackgroundColor3 = THEME.bad })
		task.delay(2, function()
			if clearArmed then
				clearArmed = false
				clearBtn.Text = "Clear"
				tween(clearBtn, 0.2, { BackgroundColor3 = THEME.box })
			end
		end)
		return
	end
	clearArmed = false
	clearBtn.Text = "Clear"
	tween(clearBtn, 0.2, { BackgroundColor3 = THEME.box })

	local res = callBuild("clear")
	if res.count then blockCount = res.count end
	refreshBuildInfo()
	stackBase, stackLevel = nil, 0
	notify("All your blocks were removed", "good")
end

local function styleTabs()
	for name, b in pairs(tabButtons) do
		local selected = (name == currentPage)
		tween(b, 0.2, { TextColor3 = selected and Color3.new(1, 1, 1) or THEME.subtext })
	end

	tween(ctx.tabPill, 0.3, { Position = UDim2.fromOffset(x, 0) }, Enum.EasingStyle.Quint)
	local view = WIDTH - 32
	local maxScroll = math.max(0, ctx.tabStrip.CanvasSize.X.Offset - view)
	tween(ctx.tabStrip, 0.3, {
		CanvasPosition = Vector2.new(math.clamp(x - (view - TAB_W) / 2, 0, maxScroll), 0),
	}, Enum.EasingStyle.Quint)
end

local function showPage(name)
	if switching or name == currentPage then return end
	switching = true

	if listening == "gui" or listening == "fly" or listening == "build" then
		listening = nil
		setMsg(DEFAULT_MSG)
		refreshSettings()
	end

	local prevName = currentPage
	local from, to = pages[currentPage], pages[name]
	currentPage = name
	styleTabs()
	local dir = (ctx.tabIndex[name] > ctx.tabIndex[prevName]) and 1 or -1

	-- if the new page is taller, slide the window up so nothing falls off the bottom of the screen
	local gh = gui.AbsoluteSize.Y
	local newH = mainHeight(name) * userScale
	local top = main.AbsolutePosition.Y
	if gh > 200 and top + newH > gh - 8 then
		local dy = (math.max(8, gh - 8 - newH) - top) / userScale
		tween(main, 0.3, { Position = main.Position + UDim2.fromOffset(0, dy) }, Enum.EasingStyle.Quint)
	end

	tween(main, 0.3, { Size = UDim2.fromOffset(WIDTH, mainHeight(name)) }, Enum.EasingStyle.Quint)
	-- old page slides + fades out, new page slides + fades in from the other side
	tween(from, 0.14, { GroupTransparency = 1, Position = UDim2.fromOffset(-dir * 18, PAGE_TOP) }).Completed:Wait()
	from.Visible = false
	from.Position = UDim2.fromOffset(0, PAGE_TOP)
	to.Position = UDim2.fromOffset(dir * 18, PAGE_TOP)
	to.Visible = true
	tween(to, 0.26, { GroupTransparency = 0, Position = UDim2.fromOffset(0, PAGE_TOP) }, Enum.EasingStyle.Quint)

	switching = false
end

-- KEY CAPTURE + KEYBINDS

local function handleCapture(key)
	if key == Enum.KeyCode.Unknown then return end
	local which = listening

	local function reject(msg)
		if which == "setup" then
			setupMsg.TextColor3 = THEME.bad
			setupMsg.Text = msg .. " - pick another key"
		else
			if which == "build" then
				notify(msg .. " - pick another key", "bad")
			else
				setMsg(msg .. " - pick another key", THEME.bad)
			end
		end
	end

	if key == Enum.KeyCode.Escape then
		if which == "setup" then
			reject("Escape opens the Roblox menu")
		else
			listening = nil
			setMsg(DEFAULT_MSG)
			refreshSettings()
		end
		return
	end

	if RESERVED[key] then
		reject(RESERVED[key])
		return
	end

	local KEY_USE = { gui = "minimizing", fly = "flying", build = "placing blocks" }
	local KEY_TITLE = { gui = "Minimize key", fly = "Fly key", build = "Add-block key" }
	local target = (which == "setup") and "gui" or which
	for name, k in pairs(keys) do
		if name ~= target and k == key then
			reject(keyName(key) .. " is already used for " .. KEY_USE[name])
			return
		end
	end

	keys[target] = key
	listening = nil

	if which == "setup" then
		pulsing = false
		keyBox.Text = keyName(key)
		setupMsg.TextColor3 = THEME.good
		setupMsg.Text = "Bound! Opening menu..."
		tween(keyStroke, 0.2, { Color = THEME.good, Transparency = 0 })
		refreshSettings()
		finishSetup()
	else
		setMsg(
			KEY_TITLE[target] .. " set to " .. keyName(key),
			THEME.good
		)
		notify(KEY_TITLE[target] .. " is now " .. keyName(key), "good")
		refreshSettings()
	end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if listening and input.UserInputType == Enum.UserInputType.Keyboard then
		handleCapture(input.KeyCode)
		return -- the capture keypress never triggers fly / minimize
	end

	if gameProcessed or not setupDone then return end

	-- tap teleport (phones): a quick tap on the world, not a camera drag
	if clickTP and device == "mobile" and input.UserInputType == Enum.UserInputType.Touch then
		local startPos = input.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End and (input.Position - startPos).Magnitude < 10 then
				camera = workspace.CurrentCamera
				local ray = camera:ScreenPointToRay(startPos.X, startPos.Y)
				local ignore = {}
				if player.Character then table.insert(ignore, player.Character) end
				if floatPlatform then table.insert(ignore, floatPlatform) end
				local params = RaycastParams.new()
				params.FilterType = Enum.RaycastFilterType.Exclude
				params.FilterDescendantsInstances = ignore
				local hit = workspace:Raycast(ray.Origin, ray.Direction * 3000, params)
				if hit then
					teleportTo(hit.Position)
					notify("Teleported", "good")
				end
			end
		end)
		return
	end

	-- Ctrl + Click teleport
	if clickTP and input.UserInputType == Enum.UserInputType.MouseButton1
		and UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
		camera = workspace.CurrentCamera
		local loc = UserInputService:GetMouseLocation()
		local ray = camera:ViewportPointToRay(loc.X, loc.Y)
		local ignore = {}
		if player.Character then table.insert(ignore, player.Character) end
		if floatPlatform then table.insert(ignore, floatPlatform) end
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = ignore
		local hit = workspace:Raycast(ray.Origin, ray.Direction * 3000, params)
		if hit then
			teleportTo(hit.Position)
			notify("Teleported", "good")
		end
		return
	end

	if input.KeyCode == keys.fly then
		toggleFly()
	elseif input.KeyCode == keys.gui then
		setGuiOpen(not guiOpen)
	elseif input.KeyCode == keys.build then
		task.spawn(placeBlock)
	end
end)

-- PER-FRAME: rainbow accent, border shimmer, live stats
---------------------------------------------------------------------
do
	local rainbowHue, rainbowAcc, statAcc, shimmerAcc = 0, 0, 0, 0
	local shownSpeed, shownAlt = 0, 0

	RunService.Heartbeat:Connect(function(dt)
		-- rainbow accent: ~20 updates a second, and only accent-coloured things are touched
		if rainbowOn then
			rainbowAcc += dt
			if rainbowAcc >= 0.05 then
				rainbowHue = (rainbowHue + rainbowAcc * 0.12) % 1
				rainbowAcc = 0
				THEME.accent = Color3.fromHSV(rainbowHue, 0.6, 1)
				applyTheme(true, true)
			end
		end

		if guiOpen then
			-- slowly spinning shimmer on the window border (30 fps is plenty)
			shimmerAcc += dt
			if shimmerAcc >= 0.033 then
				ctx.shimmer.Rotation = (ctx.shimmer.Rotation + shimmerAcc * 55) % 360
				shimmerAcc = 0
			end

			statAcc += dt
			if statAcc >= 0.1 then
				statAcc = 0
				if currentPage == "fly" then
					local _, root = getCharacterParts()
					if root then
						-- numbers glide to their new value instead of jumping
						shownSpeed += (root.AssemblyLinearVelocity.Magnitude - shownSpeed) * 0.55
						shownAlt += (root.Position.Y - shownAlt) * 0.55
						speedVal.Text = tostring(math.floor(shownSpeed + 0.5))
						altVal.Text = tostring(math.floor(shownAlt + 0.5))
					end
				end
			end
		end
	end)
end

local function initMobileHud()
	local hud = Instance.new("Frame")
	hud.Name = "MobileHud"
	hud.BackgroundTransparency = 1
	hud.AnchorPoint = Vector2.new(1, 0.5)
	hud.Position = UDim2.new(1, -14, 0.5, 0)
	hud.Size = UDim2.fromOffset(56, 236)
	hud.Parent = gui

	local function circle(y, text, size, parent)
		local b = Instance.new("TextButton")
		b.Size = UDim2.fromOffset(56, 56)
		b.Position = UDim2.fromOffset(0, y)
		b.BorderSizePixel = 0
		b.AutoButtonColor = false
		b.Font = Enum.Font.GothamBold
		b.Text = text
		b.TextSize = size
		b.Parent = parent or hud
		round(b, 28)
		bind(b, "TextColor3", "text")
		stroke(b, "accent", 2, 0.25)
		return b
	end

	local upBtn = circle(0, "▲", 18)
	local flyBtn = circle(60, "✈", 24)
	local downBtn = circle(120, "▼", 18)
	local blockBtn = circle(180, "+", 30)
	for _, b in ipairs({ upBtn, downBtn, blockBtn }) do
		bind(b, "BackgroundColor3", "bar")
	end
	flyBtn.BackgroundColor3 = THEME.bar

	-- hold-to-move buttons
	local function hold(btn, key)
		btn.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.Touch
				or input.UserInputType == Enum.UserInputType.MouseButton1 then
				touch[key] = true
				tween(btn, 0.08, { BackgroundColor3 = THEME.accent })
				input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then
						touch[key] = false
						tween(btn, 0.15, { BackgroundColor3 = THEME.bar })
					end
				end)
			end
		end)
	end
	hold(upBtn, "up")
	hold(downBtn, "down")

	flyBtn.MouseButton1Click:Connect(toggleFly)
	blockBtn.MouseButton1Click:Connect(function() task.spawn(placeBlock) end)

	-- combat buttons (only shown while those features are on)
	local hud2 = Instance.new("Frame")
	hud2.Name = "CombatHud"
	hud2.BackgroundTransparency = 1
	hud2.AnchorPoint = Vector2.new(1, 0.5)
	hud2.Position = UDim2.new(1, -80, 0.5, 0)
	hud2.Size = UDim2.fromOffset(56, 116)
	hud2.Parent = gui
	local aimBtn = circle(0, "AIM", 13, hud2)
	local fireBtn = circle(60, "FIRE", 13, hud2)
	bind(aimBtn, "BackgroundColor3", "bar")
	bind(fireBtn, "BackgroundColor3", "bar")

	local function holdCombat(btn, key)
		btn.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.Touch
				or input.UserInputType == Enum.UserInputType.MouseButton1 then
				local cb = ctx.combat
				if key == "aim" and cb and cb.aim.mode == "Toggle" then
					-- tap to lock on / release
					cb.aimToggled = not cb.aimToggled
					notify("Aimbot " .. (cb.aimToggled and "locked on" or "released"), cb.aimToggled and "good" or "bad")
					return
				end
				if cb then cb.hold[key] = true end
				tween(btn, 0.08, { BackgroundColor3 = THEME.accent })
				input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then
						if ctx.combat then ctx.combat.hold[key] = false end
						tween(btn, 0.15, { BackgroundColor3 = THEME.bar })
					end
				end)
			end
		end)
	end
	holdCombat(aimBtn, "aim")
	holdCombat(fireBtn, "fire")

	-- up / down only matter while flying, swimming or floating
	local acc = 0
	RunService.Heartbeat:Connect(function(dt)
		acc += dt
		if acc < 0.12 then return end
		acc = 0
		local moving = mode ~= nil or floatOn
		upBtn.Visible = moving
		downBtn.Visible = moving
		if not moving then
			touch.up = false
			touch.down = false
		end
		flyBtn.BackgroundColor3 = (mode ~= nil) and THEME.accent or THEME.bar
		local cb = ctx.combat
		if cb then
			aimBtn.Visible = cb.aim.enabled or cb.silent.enabled
			fireBtn.Visible = cb.blaster.enabled
			if cb.aim.mode == "Toggle" and not cb.hold.aim then
				aimBtn.BackgroundColor3 = cb.aimToggled and THEME.accent or THEME.bar
			end
		end
	end)
end

	local function num(v, lo, hi)
		local n = tonumber(v)
		if not n then return nil end
		if lo then n = math.max(lo, n) end
		if hi then n = math.min(hi, n) end
		return n
	end

	local function onOff(arg, current)
		if arg == "on" then return true end
		if arg == "off" then return false end
		return not current
	end

	local function findPlayer(q)
		if not q then return nil end
		q = q:lower()
		if q == "me" then return player end
		local partial
		for _, p in ipairs(Players:GetPlayers()) do
			local n, d = p.Name:lower(), p.DisplayName:lower()
			if n == q or d == q then return p end
			if not partial and (n:sub(1, #q) == q or d:sub(1, #q) == q) then partial = p end
		end
		return partial
	end

	local function parseColor(tokens)
		local first = tokens[1]
		if not first then return nil end
		local named = COLOR_NAMES[first:lower()]
		if named then return named end
		local hex = first:match("^#?(%x%x%x%x%x%x)$")
		if hex then
			return C(tonumber(hex:sub(1, 2), 16), tonumber(hex:sub(3, 4), 16), tonumber(hex:sub(5, 6), 16))
		end
		local r, g, b = tonumber(tokens[1]), tonumber(tokens[2]), tonumber(tokens[3])
		if r and g and b then
			return C(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255))
		end
		return nil
	end

	addCommand({ "build" }, "build [shape] [size] [material] [color]", "Set + place in one go, e.g. build ball 6 neon red", function(a)
		local b = build()
		local sizeSet = false
		local colorTokens = {}
		for _, t in ipairs(a) do
			local lower = t:lower()
			if b.hasShape(lower) then
				b.setShape(lower)
			elseif tonumber(t) and not sizeSet and #colorTokens == 0 then
				b.setSize(tonumber(t))
				sizeSet = true
			elseif b.hasMaterial(lower) then
				b.setMaterial(lower)
			else
				colorTokens[#colorTokens + 1] = t
			end
		end
		if #colorTokens > 0 then
			local c = parseColor(colorTokens)
			if not c then return "Unknown color '" .. table.concat(colorTokens, " ") .. "'", "bad" end
			b.setColor(c)
		end
		b.place()

	---------------------------------------------------------------
	-- running commands

	local history, histPos = {}, 0
	local run

	function run(text)
		text = text:gsub("^%s+", ""):gsub("%s+$", "")
		if text == "" then return end
		if text:sub(1, 1) == PREFIX then text = text:sub(2) end

		history[#history + 1] = text
		histPos = #history + 1
		log("> " .. PREFIX .. text, "dim")

		local args = {}
		for w in text:gmatch("%S+") do args[#args + 1] = w end
		local name = table.remove(args, 1)
		if not name then return end

		local entry = commands[name:lower()]
		if not entry then
			log("Unknown command '" .. name .. "'.  Type " .. PREFIX .. "cmds", "bad")
			return
		end
		local ok, msg, kind = pcall(entry.fn, args, text)
		if not ok then
			log("Error: " .. tostring(msg), "bad")
		elseif msg ~= nil and msg ~= "" then
			log(tostring(msg), kind or "good")
		end
	end

		local t = cmdBox.Text:gsub("^%s*;?%s*", "")
		local word = t:match("^(%S*)") or ""
		local picks = {}
		if t == "" then
			chipMode = "quick"
			picks = QUICK
		elseif not t:find("%s") then
			chipMode = "suggest"
			local lower = word:lower()
			for _, e in ipairs(list) do
				for _, n in ipairs(e.names) do
					if n:sub(1, #lower) == lower then
						picks[#picks + 1] = e.name
						break
					end
				end
				if #picks >= 4 then break end
			end
		else

		end
		for i, b in ipairs(chipButtons) do
			chipCmds[i] = picks[i]
			b.Visible = (chipMode ~= "usage") and picks[i] ~= nil
			b.Text = picks[i] or ""

local function initFun()
	local Lighting = game:GetService("Lighting")

	local makeSecondary, makeChips = ctx.makeSecondary, ctx.makeChips
	local defaultGravity = workspace.Gravity

		spin = false, spinSpeed = 6, bouncy = false, rainbowBody = false,
		disco = false, aura = "None", gravity = "Earth", party = false,
	}
	local api = { sw = {}, state = F }
	ctx.fun = api

	local function caption(text, y)
		label(page, text, 16, y, 250, 14, Enum.Font.GothamBold, 10, "subtext")
	end

	---------------------------------------------------------------
	-- spin
	---------------------------------------------------------------
	local spinConn
	local function setSpin(v)
		F.spin = v
		if spinConn then spinConn:Disconnect() spinConn = nil end
		if v then
			spinConn = RunService.RenderStepped:Connect(function(dt)
				if mode ~= nil then return end -- flying has its own rotation
				local hum, root = getCharacterParts()
				if not hum or not root then return end
				hum.AutoRotate = false
				root.CFrame = root.CFrame * CFrame.Angles(0, F.spinSpeed * math.min(dt, 0.1), 0)
			end)
		else
			local hum = getCharacterParts()
			if hum and mode == nil then hum.AutoRotate = true end
		end
	end

	---------------------------------------------------------------
	-- bouncy landings
	---------------------------------------------------------------
	local bounceConn
	local function hookBounce(char)
		if bounceConn then bounceConn:Disconnect() bounceConn = nil end
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hum then return end
		bounceConn = hum.StateChanged:Connect(function(_, new)
			if F.bouncy and mode == nil and new == Enum.HumanoidStateType.Landed then
				local root = char:FindFirstChild("HumanoidRootPart")
				if root then
					local v = root.AssemblyLinearVelocity
					root.AssemblyLinearVelocity = Vector3.new(v.X, 72, v.Z)
					hum:ChangeState(Enum.HumanoidStateType.Jumping)
				end
			end
		end)
	end

	---------------------------------------------------------------
	-- rainbow body (only you see it)
	---------------------------------------------------------------
	local rbParts, rbOrig, rbConn = {}, {}, nil
	local function collectParts(char)
		rbParts = {}
		if not char then return end
		for _, d in ipairs(char:GetDescendants()) do
			if d:IsA("BasePart") and d.Name ~= "HumanoidRootPart" then
				rbParts[#rbParts + 1] = d
				if rbOrig[d] == nil then rbOrig[d] = d.Color end
			end
		end
	end
	local function setRainbowBody(v)
		F.rainbowBody = v
		if rbConn then rbConn:Disconnect() rbConn = nil end
		if v then
			collectParts(player.Character)
			local hue, acc = 0, 0
			rbConn = RunService.Heartbeat:Connect(function(dt)
				acc += dt
				if acc < 0.06 then return end -- ~16 updates a second is plenty
				hue = (hue + acc * 0.35) % 1
				acc = 0
				for i, part in ipairs(rbParts) do
					if part.Parent then
						part.Color = Color3.fromHSV((hue + i * 0.045) % 1, 0.75, 1)
					end
				end
			end)
		else
			for part, col in pairs(rbOrig) do
				if part.Parent then part.Color = col end
			end
			rbOrig, rbParts = {}, {}
		end
	end

	---------------------------------------------------------------
	-- disco lighting (only you see it)
	---------------------------------------------------------------
	local discoConn, discoFx
	local function setDisco(v)
		F.disco = v
		if discoConn then discoConn:Disconnect() discoConn = nil end
		if discoFx then
			for _, e in ipairs(discoFx) do e:Destroy() end
			discoFx = nil
		end
		if v then
			local cc = Instance.new("ColorCorrectionEffect")
			cc.Name = "FlyMenuDisco"
			cc.Saturation = 0.35
			cc.Contrast = 0.08
			cc.Parent = Lighting
			local bloom = Instance.new("BloomEffect")
			bloom.Name = "FlyMenuDiscoBloom"
			bloom.Intensity = 0.55
			bloom.Size = 22
			bloom.Threshold = 0.85
			bloom.Parent = Lighting
			discoFx = { cc, bloom }

			local hue, acc = 0, 0
			discoConn = RunService.Heartbeat:Connect(function(dt)
				acc += dt
				if acc < 0.05 then return end
				hue = (hue + acc * 0.5) % 1
				acc = 0
				cc.TintColor = Color3.fromHSV(hue, 0.35, 1)
			end)
		end
	end

	local auraObj
	local function setAura(name)
		F.aura = name
		if auraObj then auraObj:Destroy() auraObj = nil end
		local _, root = getCharacterParts()

		local acc = THEME.accent

		elseif name == "Smoke" then
			auraObj = Instance.new("Smoke")
			auraObj.Size = 6
			auraObj.Opacity = 0.35

			auraObj.Color = acc

		if not GRAVITY[name] then return false end

	---------------------------------------------------------------

	function api.launch()
		local _, root = getCharacterParts()
		if not root then return end
		if mode ~= nil then
			velocity = velocity + Vector3.new(0, 140, 0) -- shove the flight velocity upward
		elseif floatPlatform then
			floatVel = 110
		else
			local v = root.AssemblyLinearVelocity
			root.AssemblyLinearVelocity = Vector3.new(v.X, 170, v.Z)
		end
		notify("Launch!", "good")
	end

	function api.randomTp()
		local _, root = getCharacterParts()
		if not root then return end
		local others = {}
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
				others[#others + 1] = p
			end
		end
		if #others > 0 then
			local p = others[math.random(1, #others)]
			teleportTo(p.Character.HumanoidRootPart.Position + Vector3.new(3, -2.9, 0))
			notify("Teleported to " .. p.Name, "good")
		else
			teleportTo(root.Position + Vector3.new(math.random(-50, 50), 12, math.random(-50, 50)))
			notify("Nobody else here - random hop!", "good")
		end
	end

	function api.trip()
		local hum, root = getCharacterParts()
		if not hum or not root or mode ~= nil then return end
		hum.PlatformStand = true
		root.AssemblyLinearVelocity = root.CFrame.LookVector * 45 + Vector3.new(0, 28, 0)
		root.AssemblyAngularVelocity = Vector3.new(math.random(-8, 8), math.random(-3, 3), math.random(-8, 8))
		task.delay(2, function()
			local h = getCharacterParts()
			if h and mode == nil then
				h.PlatformStand = false
				h:ChangeState(Enum.HumanoidStateType.GettingUp)
			end
		end)
		notify("Whoops!", "bad")
	end

	function api.emote(name)
		local hum = getCharacterParts()
		if not hum then return false end
		local ok, played = pcall(function() return hum:PlayEmote(name) end)
		if not (ok and played) then
			notify("That emote isn't available in this game", "bad")
			return false
		end
		return true
	end

		n = n or 60

					Position = UDim2.fromOffset(x + math.random(-90, 90), gh + 30),
					Rotation = f.Rotation + math.random(-540, 540),
				}, Enum.EasingStyle.Sine)

			end)

	end

	---------------------------------------------------------------
	-- shared helpers
	---------------------------------------------------------------
	local function aliveParts(p)
		local char = p.Character
		if not char then return end
		local hum = char:FindFirstChildOfClass("Humanoid")
		local root = char:FindFirstChild("HumanoidRootPart")
		if not hum or not root or hum.Health <= 0 then return end
		return char, hum, root
	end

	local function sameTeam(p)
		return player.Team ~= nil and p.Team == player.Team
	end

	local function torsoOf(char)
		return char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
	end

	local function partsFor(char, which)
		local list = {}
		if which ~= "Torso" then
			local h = char:FindFirstChild("Head")
			if h then list[#list + 1] = h end
		end
		if which ~= "Head" or #list == 0 then
			local t = torsoOf(char)
			if t then list[#list + 1] = t end
		end
		return list
	end

	-- where the crosshair is: the mouse, or the middle of the screen when the camera is locked / on a phone
	local function aimCenter()
		if UIS.MouseBehavior == Enum.MouseBehavior.LockCenter or device == "mobile" then
			local vs = workspace.CurrentCamera.ViewportSize
			return Vector2.new(vs.X / 2, vs.Y / 2)
		end
		return UIS:GetMouseLocation()
	end

	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	local ignoreList = {}
	local function setIgnore()
		ignoreList[1] = player.Character
		ignoreList[2] = floatPlatform
		rayParams.FilterDescendantsInstances = ignoreList
	end

	local function visible(fromPos, part, char)
		setIgnore()
		local hit = workspace:Raycast(fromPos, part.Position - fromPos, rayParams)
		return hit == nil or hit.Instance:IsDescendantOf(char)
	end

	---------------------------------------------------------------
	-- target selection (used by aimbot + silent aim)
	---------------------------------------------------------------
	local function evaluate(p, cfg, fovRadius, cam, origin, center)
		if p == player or (cfg.teamCheck and sameTeam(p)) then return nil end
		local char, hum = aliveParts(p)
		if not char then return nil end

		local best, bestD
		for _, part in ipairs(partsFor(char, cfg.part)) do
			local pos = part.Position + part.AssemblyLinearVelocity * cfg.prediction
			local sp, onScreen = cam:WorldToViewportPoint(pos)
			if onScreen then
				local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
				if d2 <= fovRadius then
					local d3 = (pos - origin).Magnitude
					if d3 <= cfg.maxDist and (not cfg.wallCheck or visible(origin, part, char)) then
						if not bestD or d2 < bestD then
							bestD = d2
							best = { player = p, char = char, hum = hum, part = part, pos = pos, dist = d3, screenDist = d2 }
						end
					end
				end
			end
		end
		return best
	end

	local function getTarget(cfg, fovRadius)
		local cam = workspace.CurrentCamera
		local origin, center = cam.CFrame.Position, aimCenter()
		local best, bestScore
		for _, p in ipairs(Players:GetPlayers()) do
			local t = evaluate(p, cfg, fovRadius, cam, origin, center)
			if t then
				local score = t.screenDist
				if cfg.priority == "Distance" then score = t.dist
				elseif cfg.priority == "Health" then score = t.hum.Health end
				if not bestScore or score < bestScore then
					best, bestScore = t, score
				end
			end
		end
		return best
	end

	---------------------------------------------------------------
	-- silent aim: redirect a shot toward the target without touching the camera
	---------------------------------------------------------------
	local lastSilent
	local function redirect(origin, direction)
		local s = CB.silent
		if not s.enabled or not cheatsAllowed() or math.random() * 100 > s.hitChance then
			return direction, nil
		end
		local t = getTarget(s, s.fov)
		lastSilent = t
		if t then
			return (t.pos - origin).Unit, t
		end
		return direction, nil
	end

	-- your own gun scripts can call this too:  direction = shared.FlyMenuAim.redirect(origin, direction)
	shared.FlyMenuAim = {
		redirect = redirect,
		getTarget = function() return getTarget(CB.silent, CB.silent.fov) end,

	}

	local markerToken = 0
	local function hitMarker(headshot, killed)
		markerToken += 1
		local my = markerToken
		local col = killed and Color3.fromRGB(255, 70, 70) or (headshot and Color3.fromRGB(255, 215, 70) or Color3.new(1, 1, 1))
		for _, b in ipairs(markerBars) do
			b.BackgroundColor3 = col
			b.BackgroundTransparency = 0
			tween(b, 0.3, { BackgroundTransparency = 1 })
		end
		marker.Visible = true
		task.delay(0.32, function()
			if my == markerToken then marker.Visible = false end
		end)
	end

	local function tracer(a, b)
		local len = (b - a).Magnitude
		if len < 1 then return end
		local p = Instance.new("Part")
		p.Anchored = true
		p.CanCollide = false
		p.CanQuery = false
		p.CanTouch = false
		p.Material = Enum.Material.Neon
		p.Color = THEME.accent
		p.Size = Vector3.new(0.12, 0.12, len)
		p.CFrame = CFrame.lookAt(a, b) * CFrame.new(0, 0, -len / 2)
		p.Parent = workspace
		Debris:AddItem(p, 0.07)
	end

	local shotAcc = 0
	local function fireOnce()
		local cam = workspace.CurrentCamera
		local char, _, root = aliveParts(player)
		if not char then return end

		local center = aimCenter()
		local ray = cam:ViewportPointToRay(center.X, center.Y)
		local origin, dir = ray.Origin, ray.Direction
		dir = redirect(origin, dir) -- <- silent aim happens here

		setIgnore()
		local hit = workspace:Raycast(origin, dir * 50000, rayParams)
		local endPos = hit and hit.Position or (origin + dir * 300)
		local head = char:FindFirstChild("Head")
		local muzzle = (head and head.Position or root.Position) + dir * 1.5
		tracer(muzzle, endPos)

		if shootRemote then shootRemote:FireServer(origin, dir) end
	end

	local function enemyUnderCrosshair()
		local cam = workspace.CurrentCamera
		local center = aimCenter()
		local ray = cam:ViewportPointToRay(center.X, center.Y)
		setIgnore()
		local hit = workspace:Raycast(ray.Origin, ray.Direction * 50000, rayParams)
		if not hit then return false end
		local model = hit.Instance:FindFirstAncestorOfClass("Model")
		local p = model and Players:GetPlayerFromCharacter(model)
		return p ~= nil and not sameTeam(p)
	end

	RunService.Heartbeat:Connect(function(dt)
		local b = CB.blaster
		if not b.enabled then
			shotAcc = 0
			return
		end
		if CB.hold.fire or (b.trigger and enemyUnderCrosshair()) then
			-- every shot the fire rate allows this frame (so rates above the frame rate work)
			shotAcc = math.min(shotAcc + dt * b.rate, b.rate * 0.25 + 1)
			local shots = 0
			while shotAcc >= 1 and shots < 30 do
				shotAcc -= 1
				shots += 1
				fireOnce()
			end
		else
			shotAcc = math.min(shotAcc, 1)
		end
	end)

	UIS.InputBegan:Connect(function(input, gp)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			if not gp and setupDone and CB.blaster.enabled then CB.hold.fire = true end
		elseif input.KeyCode == Enum.KeyCode.G and not gp and setupDone and CB.aim.enabled and CB.aim.mode == "Toggle" then
			CB.aimToggled = not CB.aimToggled
			notify("Aimbot " .. (CB.aimToggled and "locked on" or "released"), CB.aimToggled and "good" or "bad")
		end
	end)
	UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then CB.hold.fire = false end
	end)

	---------------------------------------------------------------
	-- aimbot (camera lock)
	---------------------------------------------------------------
	local function aimActive()
		local a = CB.aim
		if not a.enabled or not cheatsAllowed() then return false end
		if a.mode == "Always" then return true end
		if a.mode == "Hold" then
			-- PC: hold right mouse.  Phone: hold the AIM button.
			return UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) or CB.hold.aim
		end
		return CB.aimToggled -- PC: press G.  Phone: tap the AIM button.
	end

	local aimTarget, statusLbl
	local statusAcc, retargetAcc = 0, 1

	pcall(function() RunService:UnbindFromRenderStep("FlyMenuAimbot") end)
	RunService:BindToRenderStep("FlyMenuAimbot", Enum.RenderPriority.Camera.Value + 1, function(dt)
		local a, s = CB.aim, CB.silent
		local center = aimCenter()

		-- FOV circles
		local showA = a.enabled and a.showFov
		aimCircle.Visible = showA
		if showA then
			aimCircle.Position = UDim2.fromOffset(center.X, center.Y)
			aimCircle.Size = UDim2.fromOffset(a.fov * 2, a.fov * 2)
		end
		local showS = s.enabled and s.showFov
		silentCircle.Visible = showS
		if showS then
			silentCircle.Position = UDim2.fromOffset(center.X, center.Y)
			silentCircle.Size = UDim2.fromOffset(s.fov * 2, s.fov * 2)
		end

		-- the lock itself
		if aimActive() then
			local cam = workspace.CurrentCamera
			local t = aimTarget

			-- the expensive search (raycasts, projections) runs ~30 times a second...
			retargetAcc += dt
			if retargetAcc >= 0.033 or not t then
				retargetAcc = 0
				t = nil
				if a.sticky and aimTarget then
					t = evaluate(aimTarget.player, a, math.huge, cam, cam.CFrame.Position, center)
				end
				if not t then t = getTarget(a, a.fov) end
			elseif t.part.Parent then
				-- ...but the aim point follows the target every single frame
				t.pos = t.part.Position + t.part.AssemblyLinearVelocity * a.prediction
			else
				t = nil
			end

			aimTarget = t
			if t then
				local goal = CFrame.lookAt(cam.CFrame.Position, t.pos)
				local alpha = 1
				if a.smooth < 100 then alpha = 1 - (1 - a.smooth / 100) ^ (math.min(dt, 0.1) * 60) end
				cam.CFrame = cam.CFrame:Lerp(goal, alpha)
			end
		else
			aimTarget = nil
		end

		-- status text (5 times a second)
		statusAcc += dt
		if statusAcc >= 0.2 and statusLbl and currentPage == "aim" and guiOpen then
			statusAcc = 0
			local t = aimTarget or (s.enabled and lastSilent)
			if t and t.player then
				statusLbl.Text = string.format("Target: %s  %d studs", t.player.Name, math.floor(t.dist + 0.5))
			else
				statusLbl.Text = "Target: none"
			end
		end
	end)

		local cam = workspace.CurrentCamera

		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player then
				local o = espObjs[p]

				if show then
					dist = (root.Position - camPos).Magnitude
					show = dist <= e.maxDist
				end

				if not show then
					if o then hideEsp(o) end
				else
					if not o then
						o = makeEsp()
						espObjs[p] = o
					end
					local color = espColorFor(p)

					-- chams (Highlight)
					if e.chams then
						if not o.hl then
							o.hl = Instance.new("Highlight")
							o.hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
						end
						if o.hl.Parent ~= char then o.hl.Parent = char end
						o.hl.Enabled = true
						o.hl.FillColor = color
						o.hl.OutlineColor = color
						o.hl.FillTransparency = e.fillTransparency / 100
						o.hl.OutlineTransparency = 0
					elseif o.hl then
						o.hl.Enabled = false
					end

					-- 2D box + extras
					local top = cam:WorldToViewportPoint(root.Position + Vector3.new(0, 3, 0))
					local bot = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3.5, 0))
					if top.Z > 0 and bot.Z > 0 then
						local h = math.abs(bot.Y - top.Y)
						local w = h * 0.55
						local cx = top.X

						o.box.Visible = e.boxes
						if e.boxes then
							o.box.Position = UDim2.fromOffset(cx - w / 2, top.Y)
							o.box.Size = UDim2.fromOffset(w, h)
							o.stroke.Color = color
							o.stroke.Thickness = e.thickness
						end

						o.name.Visible = e.names
						if e.names then
							o.name.Position = UDim2.fromOffset(cx, top.Y - 2)
							o.name.Text = p.DisplayName
							o.name.TextColor3 = color
						end

						o.dist.Visible = e.dist
						if e.dist then
							o.dist.Position = UDim2.fromOffset(cx, bot.Y + 2)
							o.dist.Text = string.format("[%dm]", math.floor(dist + 0.5))
						end

						o.hpBg.Visible = e.health
						if e.health then
							local ratio = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
							o.hpBg.Position = UDim2.fromOffset(cx - w / 2 - 6, top.Y)
							o.hpBg.Size = UDim2.fromOffset(3, h)
							o.hpFill.Size = UDim2.fromScale(1, ratio)
							o.hpFill.BackgroundColor3 = Color3.fromHSV(ratio * 0.33, 0.85, 1)
						end

						o.tracer.Visible = e.tracers
						if e.tracers then
							local from = Vector2.new(vs.X / 2, vs.Y)
							local to = Vector2.new(cx, bot.Y)
							local delta = to - from
							o.tracer.Position = UDim2.fromOffset((from.X + to.X) / 2, (from.Y + to.Y) / 2)
							o.tracer.Size = UDim2.fromOffset(delta.Magnitude, 1.5)
							o.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
							o.tracer.BackgroundColor3 = color
						end
					else
						o.box.Visible = false
						o.name.Visible = false
						o.dist.Visible = false
						o.hpBg.Visible = false
						o.tracer.Visible = false
					end
				end
			end
		end
	end)

	Players.PlayerRemoving:Connect(function(p)
		local o = espObjs[p]
		if o then
			for _, v in pairs(o) do
				if typeof(v) == "Instance" then v:Destroy() end
			end
			espObjs[p] = nil
		end
	end)

	---------------------------------------------------------------
	-- hitbox expander: server does the real work, this draws it + sends the settings
	---------------------------------------------------------------
	local syncPending = false
	local function syncHitbox()
		if syncPending then return end
		syncPending = true
		task.delay(0.15, function()
			syncPending = false
			if hitboxRemote then
				hitboxRemote:FireServer({ enabled = CB.hit.enabled and cheatsAllowed(), size = CB.hit.size, part = CB.hit.part })
			end
		end)
	end

	local vizObjs = {} -- [player] = { Head = sphere, Torso = sphere }
	local vizAcc = 0
	local function clearViz(p)
		local v = vizObjs[p]
		if v then
			for _, s in pairs(v) do s:Destroy() end
			vizObjs[p] = nil
		end
	end

	RunService.Heartbeat:Connect(function(dt)
		vizAcc += dt
		if vizAcc < 0.1 then return end
		vizAcc = 0

		local h = CB.hit
		local active = h.enabled and h.show and cheatsAllowed()
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player then
				local char = active and aliveParts(p) and p.Character
				if not char or (h.teamCheck and sameTeam(p)) then
					clearViz(p)
				else
					local v = vizObjs[p]
					if not v then
						v = {}
						vizObjs[p] = v
					end
					local want = {}
					if h.part ~= "Torso" then want.Head = char:FindFirstChild("Head") end
					if h.part ~= "Head" then want.Torso = torsoOf(char) end
					for name in pairs(BASE_RADIUS) do
						local part = want[name]
						local s = v[name]
						if not part then
							if s then s:Destroy() v[name] = nil end
						else
							if not s then
								s = Instance.new("SphereHandleAdornment")
								s.AlwaysOnTop = true
								s.ZIndex = 1
								s.Transparency = 0.75
								s.Parent = workspace.CurrentCamera
								v[name] = s
							end
							s.Adornee = part
							s.Radius = BASE_RADIUS[name] + h.size
							s.Color3 = THEME.accent
						end
					end
				end
			end
		end
	end)

	Players.PlayerRemoving:Connect(clearViz)
